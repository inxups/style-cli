// This probe records observed behavior; exit 0 does not imply spec compliance.
// Copy to a temporary main.go and run from the repository root as documented.
package main

import (
	"encoding/json"
	"fmt"
	"os"
	"path/filepath"

	"github.com/inxups/style-cli/core/spec"
	"github.com/inxups/style-cli/core/when"
)

type scenario struct {
	name, manifest, singleFile string
	files                      map[string]string
	checkJSON, checkWhen       bool
}

type diagnostic struct {
	Code     string `json:"code"`
	Path     string `json:"path"`
	Message  string `json:"message"`
	Severity string `json:"severity"`
}

type observation struct {
	Case               string       `json:"case"`
	LoadError          string       `json:"load_error,omitempty"`
	LoadCode           string       `json:"load_code,omitempty"`
	PackageName        string       `json:"package_name,omitempty"`
	RuleIDs            []string     `json:"rule_ids,omitempty"`
	Diagnostics        []diagnostic `json:"diagnostics"`
	EffectiveJSONError string       `json:"effective_json_error,omitempty"`
	NormalizeError     string       `json:"normalize_error,omitempty"`
}

func manifest(fields, rules string) string {
	return "spec: style/v0.3\nname: audit\nversion: 1.0.0\ndescription: Review probe\n" + fields + "rules:\n" + rules
}

func main() {
	baseRule := "  - id: base\n    target: system\n    text: body\n"
	cases := []scenario{
		{name: "R01 second YAML document", manifest: manifest("", baseRule) + "---\nname: unseen\nrules: malformed\n"},
		{name: "R01 malformed trailing YAML", manifest: manifest("", baseRule) + "---\nname: [unterminated\n"},
		{name: "R02 explicit non-Markdown from", manifest: manifest("", "  - from: rules/test.txt\n"), files: map[string]string{"rules/test.txt": "---\ntarget: system\n---\nbody\n"}},
		{name: "R02 glob non-Markdown from", manifest: manifest("", "  - from: rules/*.txt\n"), files: map[string]string{"rules/test.txt": "---\ntarget: system\n---\nbody\n"}},
		{name: "R03 NaN priority", manifest: manifest("priority: .nan\n", baseRule), checkJSON: true},
		{name: "R03 infinite budget", manifest: manifest("budget:\n  per_rule: .inf\n", baseRule), checkJSON: true},
		{name: "R04 non-string vars key", manifest: manifest("vars:\n  1: value\n", baseRule)},
		{name: "R04 non-string reads key", manifest: manifest("reads:\n  1: true\n", baseRule)},
		{name: "R05 native Edit with tool.added", manifest: manifest("", "  - id: base\n    target: tool\n    names: Edit\n    text: body\n    when:\n      conditions:\n        - {field: tool.added, op: contains, pattern: TODO}\n")},
		{name: "R06 skill before with tool.output", manifest: manifest("", "  - id: base\n    target: skill\n    skill: review\n    at: before\n    text: body\n    when:\n      conditions:\n        - {field: tool.output, op: contains, pattern: marker}\n")},
		{name: "R07 malformed model glob", manifest: manifest("reads:\n  model: true\n", "  - id: base\n    target: system\n    text: body\n    when:\n      model: '['\n"), checkWhen: true},
		{name: "R08 directory native rule", manifest: manifest("", "  - id: native\n    target: native\n    codex: {}\n    text: body\n")},
		{name: "R08 single-file native rule", singleFile: "---\nspec: style/v0.3\nname: native\nversion: 1.0.0\ndescription: Review probe\ntarget: native\ncodex: {}\n---\nbody\n"},
	}
	results := make([]observation, 0, len(cases))
	for _, c := range cases {
		result, err := observe(c)
		if err != nil {
			fmt.Fprintln(os.Stderr, err)
			os.Exit(1)
		}
		results = append(results, result)
	}
	encoder := json.NewEncoder(os.Stdout)
	encoder.SetIndent("", "  ")
	if err := encoder.Encode(results); err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(1)
	}
}

func observe(c scenario) (observation, error) {
	result := observation{Case: c.name, Diagnostics: []diagnostic{}}
	root, err := os.MkdirTemp("", "style-cli-review-case-")
	if err != nil {
		return result, err
	}
	defer os.RemoveAll(root)
	source := root
	files := map[string]string{"STYLE.yaml": c.manifest}
	if c.singleFile != "" {
		files = map[string]string{"native.style.md": c.singleFile}
		source = filepath.Join(root, "native.style.md")
	}
	for name, body := range c.files {
		files[name] = body
	}
	for name, body := range files {
		destination := filepath.Join(root, name)
		if err := os.MkdirAll(filepath.Dir(destination), 0700); err != nil {
			return result, err
		}
		if err := os.WriteFile(destination, []byte(body), 0600); err != nil {
			return result, err
		}
	}
	pkg, err := spec.Load(source)
	if err != nil {
		result.LoadError, result.LoadCode = err.Error(), spec.ErrorCode(err)
		return result, nil
	}
	result.PackageName, _ = pkg.Metadata["name"].(string)
	for _, rule := range pkg.Rules {
		id, _ := rule.Fields["id"].(string)
		result.RuleIDs = append(result.RuleIDs, id)
	}
	for _, d := range spec.Validate(pkg) {
		result.Diagnostics = append(result.Diagnostics, diagnostic{d.Code, d.Path, d.Message, d.Severity})
	}
	if c.checkJSON {
		merged, err := spec.Merge(pkg)
		if err != nil {
			return result, err
		}
		if _, err := json.Marshal(merged.Rules[0].Effective); err != nil {
			result.EffectiveJSONError = err.Error()
		}
	}
	if c.checkWhen {
		fields, _ := pkg.Rules[0].Fields["when"].(map[string]any)
		if _, err := when.Normalize(fields, "session.start"); err != nil {
			result.NormalizeError = err.Error()
		}
	}
	return result, nil
}
